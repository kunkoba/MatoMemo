# Step 1: サーバーに Docker と Docker Compose をインストール

# 1. 必要なパッケージのインストール
apt update && apt install -y ca-certificates curl gnupg

# 2. Docker公式のGPGキーを追加
install -m 0755 -d /etc/apt/keyrings
curl -fsSL https://download.docker.com/linux/ubuntu/gpg | gpg --dearmor -o /etc/apt/keyrings/docker.gpg
chmod a+r /etc/apt/keyrings/docker.gpg

# 3. リポジトリの登録
echo \
  "deb [arch=$(dpkg --print-architecture) signed-by=/etc/apt/keyrings/docker.gpg] https://download.docker.com/linux/ubuntu \
  $(. /etc/os-release && echo "$VERSION_CODENAME") stable" | \
  tee /etc/apt/sources.list.d/docker.list > /dev/null

# 4. Dockerパッケージのインストール
apt update
apt install -y docker-ce docker-ce-cli containerd.io docker-buildx-plugin docker-compose-plugin

# バージョン確認
docker compose version


■■■■■
# Step 2: サーバーに アップロード（ローカルで実行）

1. ローカルで不要なファイルを除外して zip（tar）にまとめる
tar --exclude="bin" --exclude="obj" --exclude="node_modules" --exclude=".git" --exclude=".vs" -czvf app.tar.gz -C "C:\Users\Owner\Desktop\__Developer\MatoMemo\_src" .

2. 作成した圧縮ファイルだけをサーバーに転送する
scp app.tar.gz root@163.44.124.155:/root/mato_memo/


■■■■■
3. サーバー側で解凍する（ConoHa サーバーの画面で実行）
cd /root/mato_memo

rm app.tar.gz

4. Docker コンテナのビルドと起動
cd /root/mato_memo
docker compose up -d --build
